## Условие:

Есть изображение размером M * N, в котором M * N пикселей. \
Даны координаты одного из пикселей (sr, sc), начиная от которого нужно \
закрасить все ***подходящие***.

Три условия, что пиксель будет ***подходящим***:
1. Изначальный цвет тот же, что и цвет начального пикселя.
2. Пиксель расположен вертикально или горизонтально рядом с текущим.

## Подход: 

Алгоритм поиска в ширину (bfs - breath-first search)

```c++
class Solution {
public:
    vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color) {
        const int m = static_cast<int>(image.size());
        const int n = static_cast<int>(image[0].size());

        int target_color = image[sr][sc];
        if (color == target_color) {return image;}

        std::queue<std::pair<int, int>> q;
        q.push({sr, sc});

        while (!q.empty()) {
            std::pair<int, int> cur_pos = q.front();
            if (!(
                cur_pos.first >= 0 && cur_pos.first < m &&
                cur_pos.second >= 0 && cur_pos.second < n &&
                image[cur_pos.first][cur_pos.second] == target_color
            )) {q.pop(); continue;}

            q.push({cur_pos.first - 1, cur_pos.second});
            q.push({cur_pos.first + 1, cur_pos.second});
            q.push({cur_pos.first, cur_pos.second - 1});
            q.push({cur_pos.first, cur_pos.second + 1});

            image[cur_pos.first][cur_pos.second] = color;
            q.pop();     
        }

        return image;
    }
};
```
